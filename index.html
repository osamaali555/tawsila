// ===================================================================== // الخطوة 1: امسح الدالة القديمة verifyIdCardLocallyWithOCR بالكامل // وحط مكانها الكود ده (كله) في نفس المكان (القسم 9) // ===================================================================== const EGY_GOV_CODES = new Set([ '01','02','03','04','11','12','13','14','15','16','17','18','19', '21','22','23','24','25','26','27','28','29','31','32','33','34','35','88' ]); function toWesternDigits(str) { return String(str || '') .replace(/[٠-٩]/g, d => '٠١٢٣٤٥٦٧٨٩'.indexOf(d)) .replace(/[۰-۹]/g, d => '۰۱۲۳۴۵۶۷۸۹'.indexOf(d)); } // يتحقق إن الرقم القومي سليم هيكلياً: 14 رقم، قرن، تاريخ ميلاد، كود محافظة function validateEgyptianNationalId(id) { if (!/^[23]\d{13}$/.test(id)) return null; const century = id[0] === '2' ? 1900 : 2000; const year = century + Number(id.slice(1, 3)); const month = Number(id.slice(3, 5)); const day = Number(id.slice(5, 7)); const gov = id.slice(7, 9); const birth = new Date(year, month - 1, day); const validDate = birth.getFullYear() === year && birth.getMonth() === month - 1 && birth.getDate() === day && birth <= new Date(); if (!validDate) return null; if (!EGY_GOV_CODES.has(gov)) return null; const now = new Date(); let age = now.getFullYear() - year; if (now < new Date(now.getFullYear(), month - 1, day)) age--; return { nationalId: id, birthDate: birth, age, gender: Number(id[12]) % 2 === 1 ? 'male' : 'female' }; } function findEgyptianNationalId(text) { // يسمح بمسافات/شرطات بين الأرقام لأن OCR ساعات بيفصلها const candidates = text.match(/[23](?:[\s.\-]?\d){13}/g) || []; for (const c of candidates) { const parsed = validateEgyptianNationalId(c.replace(/\D/g, '')); if (parsed) return parsed; } return null; } async function verifyIdCardLocallyWithOCR(ocrImageUrl, onProgress, imageAspect) { // فحص نسبة الأبعاد: البطاقة أفقية (~1.58). السيلفي والصور الطولية بتترفض هنا. if (typeof imageAspect === 'number' && (imageAspect < 1.25 || imageAspect > 2.1)) { return { isIdCard: false, reason: 'شكل الصورة لا يطابق البطاقة الشخصية. صوّر البطاقة بشكل أفقي وبحيث تملأ الصورة.' }; } const TesseractLib = await ensureTesseractLoaded(); const result = await TesseractLib.recognize(ocrImageUrl, 'ara+eng', { logger: (m) => { if (m.status === 'recognizing text' && typeof onProgress === 'function') { onProgress(Math.round((m.progress || 0) * 100)); } } }); const rawText = result?.data?.text || ''; const westernText = toWesternDigits(rawText); const normalized = westernText .replace(/[أإآ]/g, 'ا') .replace(/ة/g, 'ه') .replace(/ى/g, 'ي') .replace(/\s+/g, ' ') .trim(); // ---------- نظام النقاط ---------- let score = 0; // 1) الرقم القومي السليم (أقوى دليل) = 3 نقاط const idInfo = findEgyptianNationalId(westernText); if (idInfo) score += 3; // 2) عبارات رسمية مميزة = نقطتين لكل عبارة (بحد أقصى 4) const phrases = ['جمهوريه مصر', 'مصر العربيه', 'بطاقه تحقيق', 'تحقيق الشخصيه', 'الرقم القومي']; const phraseHits = phrases.filter(p => normalized.includes(p)).length; score += Math.min(phraseHits, 2) * 2; // 3) كلمات مفردة طويلة = نقطة لكل كلمة (بحد أقصى 3) const words = ['جمهوريه', 'العربيه', 'بطاقه', 'تحقيق', 'الشخصيه', 'القومي', 'مسلم', 'مسيحي', 'الديانه', 'الوظيفه']; const wordHits = words.filter(w => normalized.includes(w)).length; score += Math.min(wordHits, 3); // 4) سيريال المصنع (حرفين + 7 أرقام) = نقطة const compact = westernText.replace(/[^A-Za-z0-9]/g, ''); if (/[A-Z]{2}\d{7}/.test(compact)) score += 1; // القبول: لازم 3 نقاط على الأقل (رقم قومي سليم لوحده كفاية، // أو عبارتين رسميتين، أو عبارة + كلمتين... إلخ) if (score < 3) { return { isIdCard: false, reason: 'لم نتمكن من التعرف على بطاقة شخصية مصرية في الصورة. صوّر الوجه الأمامي للبطاقة بإضاءة جيدة ومن غير انعكاس.' }; } // لو لقينا الرقم القومي، نتأكد من السن (18+) if (idInfo && idInfo.age < 18) { return { isIdCard: false, reason: 'يجب أن يكون عمرك 18 سنة أو أكثر لاستخدام التطبيق.' }; } return { isIdCard: true, reason: 'تم التحقق من البطاقة الشخصية', nationalId: idInfo?.nationalId || null, age: idInfo?.age ?? null, score };
